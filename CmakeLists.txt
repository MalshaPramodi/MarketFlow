cmake_minimum_required(VERSION 3.20)

project(
    MarketFlow
    VERSION 1.0.0
    DESCRIPTION "C++20 Financial Market Order Matching Engine"
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(CTest)

# Core business logic library
add_library(marketflow_core
    src/models/order.cpp
)

target_include_directories(marketflow_core
    PUBLIC
        ${PROJECT_SOURCE_DIR}/include
)

# Main application
add_executable(MarketFlow
    src/main.cpp
)

target_link_libraries(MarketFlow
    PRIVATE marketflow_core
)

# Compiler warnings
if(MSVC)
    target_compile_options(marketflow_core PRIVATE /W4)
    target_compile_options(MarketFlow PRIVATE /W4)
else()
    target_compile_options(marketflow_core PRIVATE
        -Wall -Wextra -Wpedantic -Wshadow
    )

    target_compile_options(MarketFlow PRIVATE
        -Wall -Wextra -Wpedantic -Wshadow
    )
endif()

# GoogleTest
if(BUILD_TESTING)
    include(FetchContent)

    FetchContent_Declare(
        googletest
        GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG v1.15.2
    )

    FetchContent_MakeAvailable(googletest)

    add_executable(MarketFlowTests
        tests/sanity_tests.cpp
        tests/order_tests.cpp
    )

    target_link_libraries(MarketFlowTests
        PRIVATE
            marketflow_core
            GTest::gtest_main
    )

    target_compile_options(MarketFlowTests PRIVATE
        -Wall -Wextra -Wpedantic -Wshadow
    )

    include(GoogleTest)
    gtest_discover_tests(MarketFlowTests)
endif()
